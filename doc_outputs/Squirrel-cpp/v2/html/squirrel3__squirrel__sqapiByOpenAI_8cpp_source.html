1038SQRESULT sq_bindenv(HSQUIRRELVM v,SQInteger idx)
1039{
1040 SQObjectPtr &o = stack_get(v,idx);
1041 if(!sq_isnativeclosure(o) &&
1042 !sq_isclosure(o))
1043 return sq_throwerror(v,_SC("the target is not a closure"));
1044 SQObjectPtr &env = stack_get(v,-1);
1045 if(!sq_istable(env) &&
1046 !sq_isarray(env) &&
1047 !sq_isclass(env) &&
1048 !sq_isinstance(env))
1049 return sq_throwerror(v,_SC("invalid environment"));
1050 SQWeakRef *w = _refcounted(env)->GetWeakRef(sq_type(env));
1051 SQObjectPtr ret;
1052 if(sq_isclosure(o)) {
1053 SQClosure *c = _closure(o)->Clone();
1054 __ObjRelease(c->_env);
1055 c->_env = w;
1056 __ObjAddRef(c->_env);
1057 if(_closure(o)->_base) {
1058 c->_base = _closure(o)->_base;
1059 __ObjAddRef(c->_base);
1060 }
1061 ret = c;
1062 }
1063 else { //then must be a native closure
1064 SQNativeClosure *c = _nativeclosure(o)->Clone();
1065 __ObjRelease(c->_env);
1066 c->_env = w;
1067 __ObjAddRef(c->_env);
1068 ret = c;
1069 }
1070 v->Pop();
1071 v->Push(ret);
1072 return SQ_OK;
