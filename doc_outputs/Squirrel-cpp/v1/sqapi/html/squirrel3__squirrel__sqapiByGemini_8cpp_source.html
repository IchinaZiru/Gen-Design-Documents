818SQRESULT sq_bindenv(HSQUIRRELVM v,SQInteger idx)
819{
820 SQObjectPtr &o = stack_get(v,idx);
821 if(!sq_isnativeclosure(o) &&
822 !sq_isclosure(o))
823 return sq_throwerror(v,_SC("the target is not a closure"));
824 SQObjectPtr &env = stack_get(v,-1);
825 if(!sq_istable(env) &&
826 !sq_isarray(env) &&
827 !sq_isclass(env) &&
828 !sq_isinstance(env))
829 return sq_throwerror(v,_SC("invalid environment"));
830 SQWeakRef *w = _refcounted(env)->GetWeakRef(sq_type(env));
831 SQObjectPtr ret;
832 if(sq_isclosure(o)) {
833 SQClosure *c = _closure(o)->Clone();
834 __ObjRelease(c->_env);
835 c->_env = w;
836 __ObjAddRef(c->_env);
837 if(_closure(o)->_base) {
838 c->_base = _closure(o)->_base;
839 __ObjAddRef(c->_base);
840 }
841 ret = c;
842 }
843 else { //then must be a native closure
844 SQNativeClosure *c = _nativeclosure(o)->Clone();
845 __ObjRelease(c->_env);
846 c->_env = w;
847 __ObjAddRef(c->_env);
848 ret = c;
849 }
850 v->Pop();
851 v->Push(ret);
852 return SQ_OK;
853}
