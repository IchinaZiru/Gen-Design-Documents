31static bool str2num(const SQChar *s,SQObjectPtr &res,SQInteger base)
32{
33 SQChar *end;
34 const SQChar *e = s;
35 bool iseintbase = base > 13; //to fix error converting hexadecimals with e like 56f0791e
36 bool isfloat = false;
37 SQChar c;
38 while((c = *e) != _SC('\0'))
39 {
40 if (c == _SC('.') || (!iseintbase && (c == _SC('E') || c == _SC('e')))) { //e and E is for scientific notation
41 isfloat = true;
42 break;
43 }
44 e++;
45 }
46 if(isfloat){
47 SQFloat r = SQFloat(scstrtod(s,&end));
48 if(s == end) return false;
49 res = r;
50 }
51 else{
52 SQInteger r = SQInteger(scstrtol(s,&end,(int)base));
53 if(s == end) return false;
54 res = r;
55 }
56 return true;
57}

206static SQInteger __getcallstackinfos(HSQUIRRELVM v,SQInteger level)
207{
208 SQStackInfos si;
209 SQInteger seq = 0;
210 const SQChar *name = NULL;
211
212 if (SQ_SUCCEEDED(sq_stackinfos(v, level, &si)))
213 {
214 const SQChar *fn = _SC("unknown");
215 const SQChar *src = _SC("unknown");
216 if(si.funcname)fn = si.funcname;
217 if(si.source)src = si.source;
218 sq_newtable(v);
219 sq_pushstring(v, _SC("func"), -1);
220 sq_pushstring(v, fn, -1);
221 sq_newslot(v, -3, SQFalse);
222 sq_pushstring(v, _SC("src"), -1);
223 sq_pushstring(v, src, -1);
224 sq_newslot(v, -3, SQFalse);
225 sq_pushstring(v, _SC("line"), -1);
226 sq_pushinteger(v, si.line);
227 sq_newslot(v, -3, SQFalse);
228 sq_pushstring(v, _SC("locals"), -1);
229 sq_newtable(v);
230 seq=0;
231 while ((name = sq_getlocal(v, level, seq))) {
232 sq_pushstring(v, name, -1);
233 sq_push(v, -2);
234 sq_newslot(v, -4, SQFalse);
235 sq_pop(v, 1);
236 seq++;
237 }
238 sq_newslot(v, -3, SQFalse);
239 return 1;
240 }
241
242 return 0;
243}

756{

758 SQTable *tbl = _table(o);
759 SQObjectPtr ret = SQTable::Create(_ss(v),0);
760

1008static SQInteger __map_array(SQArray *dest,SQArray *src,HSQUIRRELVM v) {
1009 SQObjectPtr temp;
1010 SQInteger size = src->Size();
1011 SQObject &closure = stack_get(v, 2);
1012 v->Push(closure);
1013
1014 SQInteger nArgs = 0;
1015 if(sq_type(closure) == OT_CLOSURE) {
1016 nArgs = _closure(closure)->_function->_nparameters;
1017 }
1018 else if (sq_type(closure) == OT_NATIVECLOSURE) {
1019 SQInteger nParamsCheck = _nativeclosure(closure)->_nparamscheck;
1020 if (nParamsCheck > 0)
1021 nArgs = nParamsCheck;
1022 else // push all params when there is no check or only minimal count set
1023 nArgs = 4;
1024 }
1025
1026 for(SQInteger n = 0; n < size; n++) {
1027 src->Get(n,temp);
1028 v->Push(src);
1029 v->Push(temp);
1030 if (nArgs >= 3)
1031 v->Push(SQObjectPtr(n));
1032 if (nArgs >= 4)
1033 v->Push(src);
1034 if(SQ_FAILED(sq_call(v,nArgs,SQTrue,SQFalse))) {
1035 return SQ_ERROR;
1036 }
1037 dest->Set(n,v->GetUp(-1));
1038 v->Pop();
1039 }
1040 v->Pop();
1041 return 0;
1042}

1231static bool _hsort_sift_down(HSQUIRRELVM v,SQArray *arr, SQInteger root, SQInteger bottom, SQInteger func)
1232{
1233 SQInteger maxChild;
1234 SQInteger done = 0;
1235 SQInteger ret;
1236 SQInteger root2;
1237 while (((root2 = root * 2) <= bottom) && (!done))
1238 {
1239 if (root2 == bottom) {
1240 maxChild = root2;
1241 }
1242 else {
1243 if(!_sort_compare(v,arr,arr->_values[root2],arr->_values[root2 + 1],func,ret))
1244 return false;
1245 if (ret > 0) {
1246 maxChild = root2;
1247 }
1248 else {
1249 maxChild = root2 + 1;
1250 }
1251 }
1252
1253 if(!_sort_compare(v,arr,arr->_values[root],arr->_values[maxChild],func,ret))
1254 return false;
1255 if (ret < 0) {
1256 if (root == maxChild) {
1257 v->Raise_Error(_SC("inconsistent compare function"));
1258 return false; // We'd be swapping ourselve. The compare function is incorrect
1259 }
1260
1261 _Swap(arr->_values[root],arr->_values[maxChild]);
1262 root = maxChild;
1263 }
1264 else {
1265 done = 1;
1266 }
1267 }
1268 return true;
